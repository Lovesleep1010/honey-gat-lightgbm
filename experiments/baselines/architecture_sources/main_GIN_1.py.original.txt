from sklearn.metrics import balanced_accuracy_score, f1_score, confusion_matrix, classification_report
import lightgbm as lgb

def seed_everything(seed=42):
    random.seed(seed)
    np.random.seed(seed)
    torch.manual_seed(seed)
    if torch.cuda.is_available():
        torch.cuda.manual_seed_all(seed)
        torch.backends.cudnn.benchmark = True


def snv(X: np.ndarray):
    """Standard Normal Variate"""
    m = X.mean(axis=1, keepdims=True)
    s = X.std(axis=1, keepdims=True) + 1e-8
    return (X - m) / s


def compute_class_weights(y_train: np.ndarray, num_classes: int = 5,
                          boost0: float = 2.0, mode: str = "effective") -> np.ndarray:
    """Same class weight logic as before"""
    counts = np.bincount(y_train, minlength=num_classes).astype(np.float32)
    counts = np.clip(counts, 1.0, None)

    if mode == "inverse":
        w = counts.sum() / (num_classes * counts)
    else:
        beta = 0.9999
        eff_num = 1.0 - np.power(beta, counts)
        w = (1.0 - beta) / eff_num

    w = w / (w.mean() + 1e-8)
    w[0] *= boost0
    w = w / (w.mean() + 1e-8)
    return w.astype(np.float32)

# 1) Graph & Feature Engineering
# =========================
def build_hybrid_edge_index(x_train: np.ndarray, top_k=12):
    """
    Build graph structure (Same as GAT version for fair comparison)
    """
    print(f"[Graph] Building Hybrid Graph on Train Set {x_train.shape} ...")
    corr = np.corrcoef(x_train.T)
    abs_corr = np.abs(corr)
    num_nodes = x_train.shape[1]
    edges = set()

    # A) Correlation edges
    for i in range(num_nodes):
        top_idx = np.argsort(abs_corr[i])[-(top_k + 1):][::-1]
        neigh = [j for j in top_idx if j != i][:top_k]
        for j in neigh:
            edges.add((i, j))
            edges.add((j, i))

    # B) Sequential edges
    for i in range(num_nodes - 1):
        edges.add((i, i + 1))
        edges.add((i + 1, i))

    # C) Self-loops
    for i in range(num_nodes):
        edges.add((i, i))

    edge_index = torch.tensor(list(edges), dtype=torch.long).t().contiguous()
    return edge_index


def spectrum_to_graph_sg3(x_128: np.ndarray, y_int: int, edge_index: torch.Tensor) -> Data:
    """SG features: s0, d1, d2"""
    s = x_128.astype(np.float32)
    win = 15
    poly = 3
    s0 = savgol_filter(s, window_length=win, polyorder=poly, deriv=0).astype(np.float32)
    d1 = savgol_filter(s, window_length=win, polyorder=poly, deriv=1).astype(np.float32)
    d2 = savgol_filter(s, window_length=win, polyorder=poly, deriv=2).astype(np.float32)
    node_feat = np.stack([s0, d1 * 10.0, d2 * 100.0], axis=1).astype(np.float32)

    return Data(x=torch.from_numpy(node_feat), edge_index=edge_index, y=torch.tensor(y_int, dtype=torch.long))


# =========================
# 2) GIN Encoder (The New Part)
# =========================
class GINEncoder(nn.Module):
    def __init__(self, in_dim=3, hid=64, dropout=0.5, num_classes=5):
        super().__init__()
        self.dropout = dropout

        # GIN 需要定义内部的 MLP
        # Layer 1 MLP
        self.mlp1 = nn.Sequential(
            nn.Linear(in_dim, hid),
            nn.BatchNorm1d(hid),
            nn.ReLU(),
            nn.Linear(hid, hid),
            nn.BatchNorm1d(hid),
            nn.ReLU(),
        )
        # GINConv 1
        self.gin1 = GINConv(self.mlp1, train_eps=True)  # train_eps=True 让模型自己学习中心节点权重

        # Layer 2 MLP
        self.mlp2 = nn.Sequential(
            nn.Linear(hid, hid),
            nn.BatchNorm1d(hid),
            nn.ReLU(),
            nn.Linear(hid, hid),
            nn.BatchNorm1d(hid),
            nn.ReLU(),
        )
        # GINConv 2
        self.gin2 = GINConv(self.mlp2, train_eps=True)

        # Classification Head
        self.head = nn.Sequential(
            nn.Linear(hid * 2, hid),
            nn.ReLU(),
            nn.Dropout(dropout),
            nn.Linear(hid, num_classes)
        )

    def forward(self, data: Data):
        x, edge_index, batch = data.x, data.edge_index, data.batch

        # GIN Layer 1
        x = self.gin1(x, edge_index)
        x = F.dropout(x, p=self.dropout, training=self.training)

        # GIN Layer 2
        x = self.gin2(x, edge_index)
        x = F.dropout(x, p=self.dropout, training=self.training)

        # Pooling (Mean + Max)
        g_mean = global_mean_pool(x, batch)
        g_max = global_max_pool(x, batch)
        emb = torch.cat([g_mean, g_max], dim=1)  # [B, 2*hid]

        logits = self.head(emb)
        return emb, logits


def train_encoder_with_val(model, train_loader, val_loader, device,
                           class_weights_t: torch.Tensor, epochs=50, lr=1e-3,
                           ckpt_name="best_gin.pt"):
    optimizer = torch.optim.AdamW(model.parameters(), lr=lr, weight_decay=1e-4)
    scheduler = torch.optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=epochs)
    criterion = nn.CrossEntropyLoss(weight=class_weights_t, label_smoothing=0.1)

    best_val_acc = 0.0
    print(f"\n[Start GIN Training] (Ckpt: {ckpt_name})")

    for ep in range(1, epochs + 1):
        model.train()
        total_loss = 0.0
        for data in train_loader:
            data = data.to(device)
            optimizer.zero_grad()
            _, logits = model(data)
            loss = criterion(logits, data.y)
            loss.backward()
            optimizer.step()
            total_loss += loss.item() * data.num_graphs

        scheduler.step()

        model.eval()
        correct, total = 0, 0
        with torch.no_grad():
            for data in val_loader:
                data = data.to(device)
                _, logits = model(data)
                pred = logits.argmax(dim=1)
                correct += (pred == data.y).sum().item()
                total += data.y.size(0)

        val_acc = correct / max(total, 1)
        if val_acc > best_val_acc:
            best_val_acc = val_acc
            torch.save(model.state_dict(), ckpt_name)

        if ep % 10 == 0:
            print(f"Ep {ep:03d} | Loss {total_loss / len(train_loader.dataset):.4f} | Val Acc {val_acc:.4f}")

    print(f"[GIN Done] Best Val Acc: {best_val_acc:.4f}")


@torch.no_grad()
def extract_embeddings(model, loader, device):
    model.eval()
    embs, ys = [], []
    for data in loader:
        data = data.to(device)
        emb, _ = model(data)
        embs.append(emb.cpu().numpy())
        ys.append(data.y.cpu().numpy())
    return np.concatenate(embs, axis=0), np.concatenate(ys, axis=0)


# =========================
# 3) Main
# =========================
def main():
    seed_everything(42)

    # =============== 修改为你的路径 ===============
    PATH = r"adulteration_dataset_26_08_2021.csv"
    # ============================================

    if not os.path.exists(PATH):
        print(f"Error: File not found at {PATH}")
        return

    print(f"Loading data from {PATH}...")
    df = pd.read_csv(PATH)
    df["Concentration_Class"] = df["Concentration_Class"].astype(str)

    keep = {"0", "5", "10", "25", "50"}
    df = df[df["Concentration_Class"].isin(keep)].copy().reset_index(drop=True)

    spec_cols = sorted([c for c in df.columns if str(c).endswith("nm")], key=lambda x: float(str(x).replace("nm", "")))
    X_raw = df[spec_cols].to_numpy(np.float32)
    X = snv(X_raw)

    classes_cc = ["0", "5", "10", "25", "50"]
    cc2id = {c: i for i, c in enumerate(classes_cc)}
    y = df["Concentration_Class"].map(cc2id).to_numpy(np.int64)

    acq = df["Acquisition"].to_numpy()
    train_idx = np.where(np.isin(acq, [1, 2, 3, 4]))[0]
    val_idx = np.where(np.isin(acq, [5]))[0]
    test_idx = np.where(np.isin(acq, [6]))[0]

    y_train_only = y[train_idx]
    # 使用和 GAT 相同的加权策略，保证公平
    class_w = compute_class_weights(y_train_only, num_classes=5, boost0=2.0)
    print(f"Class Weights: {np.round(class_w, 3)}")

    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    class_weights_t = torch.tensor(class_w, dtype=torch.float32, device=device)

    EDGE_INDEX = build_hybrid_edge_index(X[train_idx], top_k=12)

    graphs_train = [spectrum_to_graph_sg3(X[i], int(y[i]), EDGE_INDEX) for i in train_idx]
    graphs_val = [spectrum_to_graph_sg3(X[i], int(y[i]), EDGE_INDEX) for i in val_idx]
    graphs_test = [spectrum_to_graph_sg3(X[i], int(y[i]), EDGE_INDEX) for i in test_idx]

    train_loader = DataLoader(graphs_train, batch_size=128, shuffle=True)
    val_loader = DataLoader(graphs_val, batch_size=256, shuffle=False)
    test_loader = DataLoader(graphs_test, batch_size=256, shuffle=False)

    # ===== Stage 1: Train GIN Encoder =====
    # GIN 比较吃参数，hid 用 64 就很强了
    model = GINEncoder(in_dim=3, hid=64, dropout=0.5, num_classes=5).to(device)

    ckpt_name = "best_gin_encoder_strict.pt"
    train_encoder_with_val(
        model, train_loader, val_loader, device,
        class_weights_t=class_weights_t,
        epochs=50, lr=1e-3, ckpt_name=ckpt_name
    )
    model.load_state_dict(torch.load(ckpt_name, map_location=device))

    # ===== Stage 2: Embeddings =====
    print("Extracting GIN embeddings...")
    Xtr_emb, ytr = extract_embeddings(model, train_loader, device)
    Xval_emb, yval = extract_embeddings(model, val_loader, device)
    Xte_emb, yte = extract_embeddings(model, test_loader, device)

    # ===== Stage 3: LightGBM =====
    lgb_class_weight = {i: float(class_w[i]) for i in range(5)}

    print("Training LightGBM on GIN features...")
    lgbm = lgb.LGBMClassifier(
        objective="multiclass", num_class=5, n_estimators=3000,
        learning_rate=0.01, num_leaves=31, max_depth=-1,
        min_child_samples=30, subsample=0.8, colsample_bytree=0.8,
        class_weight=lgb_class_weight, random_state=42, n_jobs=-1
    )

    lgbm.fit(
        Xtr_emb, ytr,
        eval_set=[(Xval_emb, yval)],
        eval_metric="multi_logloss",
        callbacks=[lgb.early_stopping(stopping_rounds=100, verbose=False)]
    )

    best_iter = lgbm.best_iteration_
    if best_iter:
        X_trainval = np.vstack([Xtr_emb, Xval_emb])
        y_trainval = np.concatenate([ytr, yval])
        lgbm_final = lgb.LGBMClassifier(
            objective="multiclass", num_class=5, n_estimators=best_iter,
            learning_rate=0.01, num_leaves=31, class_weight=lgb_class_weight,
            random_state=42, n_jobs=-1
        )
        lgbm_final.fit(X_trainval, y_trainval)
    else:
        lgbm_final = lgbm

    # ===== Stage 4: Blind Test =====
    print("\n" + "=" * 40)
    print(" [FINAL BLIND TEST] GIN + LGBM (Acq 6) ")
    print("=" * 40)

    pred = lgbm_final.predict(Xte_emb)
    bal_acc = balanced_accuracy_score(yte, pred)
    macro_f1 = f1_score(yte, pred, average="macro")

    print(f"Balanced Acc: {bal_acc:.4f}")
    print(f"Macro F1:     {macro_f1:.4f}")
    print(confusion_matrix(yte, pred))
    print(classification_report(yte, pred, target_names=classes_cc, digits=4))


if __name__ == "__main__":
    main()
