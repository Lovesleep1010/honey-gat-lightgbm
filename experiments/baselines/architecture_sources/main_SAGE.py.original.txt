# ============================================================
# Strict HSI Honey Adulteration (Acq split) with GraphSAGE:
#   - No downsampling (keep 128 bands)
#   - SNV + SG3 node features: [s0, d1, d2] per band (node dim=3)
#   - Hybrid graph (corr Top-k + sequential + self-loop) built on TRAIN only
#   - GraphSAGE encoder (2-layer) + FocalLoss (class-weighted)
#   - LightGBM classifier on graph embeddings (mean+max pooled)
#
# Split: Train=Acq1-4, Val=Acq5, Test=Acq6 (final blind)
# ============================================================

import os
import random
import warnings
import numpy as np
import pandas as pd
from scipy.signal import savgol_filter

import torch
import torch.nn as nn
import torch.nn.functional as F

from torch_geometric.data import Data
from torch_geometric.loader import DataLoader
from torch_geometric.nn import SAGEConv, global_mean_pool, global_max_pool

from sklearn.metrics import balanced_accuracy_score, f1_score, confusion_matrix, classification_report
import lightgbm as lgb

warnings.filterwarnings("ignore")
os.environ["TF_CPP_MIN_LOG_LEVEL"] = "3"


# =========================
# 0) Utils
# =========================
def seed_everything(seed=42):
    random.seed(seed)
    np.random.seed(seed)
    torch.manual_seed(seed)
    if torch.cuda.is_available():
        torch.cuda.manual_seed_all(seed)
        torch.backends.cudnn.benchmark = True


def snv(X: np.ndarray) -> np.ndarray:
    m = X.mean(axis=1, keepdims=True)
    s = X.std(axis=1, keepdims=True) + 1e-8
    return (X - m) / s


def compute_class_weights(y_train, num_classes=5, boost0=2.5):
    """Effective number of samples + boost class0."""
    counts = np.bincount(y_train, minlength=num_classes).astype(np.float32)
    counts = np.clip(counts, 1.0, None)
    beta = 0.9999
    eff_num = 1.0 - np.power(beta, counts)
    w = (1.0 - beta) / eff_num
    w = w / (w.mean() + 1e-8)
    w[0] *= boost0
    w = w / (w.mean() + 1e-8)
    return w.astype(np.float32)


class FocalLoss(nn.Module):
    def __init__(self, alpha=None, gamma=3.0, reduction="mean"):
        super().__init__()
        self.alpha = alpha
        self.gamma = gamma
        self.reduction = reduction

    def forward(self, inputs, targets):
        ce = F.cross_entropy(inputs, targets, reduction="none", weight=self.alpha)
        pt = torch.exp(-ce)
        loss = ((1.0 - pt) ** self.gamma) * ce
        if self.reduction == "mean":
            return loss.mean()
        if self.reduction == "sum":
            return loss.sum()
        return loss


# =========================
# 1) Graph & Feature Engineering
# =========================
def build_hybrid_edge_index(x_train, top_k=12):
    """
    x_train: [N_train, num_nodes] (num_nodes=128)
    """
    print(f"[Graph] Building Hybrid Graph on Train Set {x_train.shape} ...")
    corr = np.corrcoef(x_train.T)
    corr = np.nan_to_num(corr, nan=0.0)
    abs_corr = np.abs(corr)
    num_nodes = x_train.shape[1]

    edges = set()

    # Correlation edges
    for i in range(num_nodes):
        top_idx = np.argsort(abs_corr[i])[-(top_k + 1):][::-1]
        neigh = [j for j in top_idx if j != i][:top_k]
        for j in neigh:
            edges.add((i, j))
            edges.add((j, i))

    # Sequential edges
    for i in range(num_nodes - 1):
        edges.add((i, i + 1))
        edges.add((i + 1, i))

    # Self-loops
    for i in range(num_nodes):
        edges.add((i, i))

    edge_index = torch.tensor(list(edges), dtype=torch.long).t().contiguous()
    print(f"[Graph] nodes={num_nodes}, edges={edge_index.shape[1]}")
    return edge_index


def spectrum_to_graph_sg3(x_vec, y_int, edge_index):
    """
    SG3 features on 128-dim spectrum -> node feat [128,3]
    """
    s = x_vec.astype(np.float32)
    win = 11
    poly = 3
    s0 = savgol_filter(s, window_length=win, polyorder=poly, deriv=0)
    d1 = savgol_filter(s, window_length=win, polyorder=poly, deriv=1)
    d2 = savgol_filter(s, window_length=win, polyorder=poly, deriv=2)

    node_feat = np.stack([s0, d1 * 10.0, d2 * 100.0], axis=1).astype(np.float32)

    return Data(
        x=torch.from_numpy(node_feat),
        edge_index=edge_index,
        y=torch.tensor(y_int, dtype=torch.long),
    )


# =========================
# 2) GraphSAGE Encoder
# =========================
class SAGEEncoder(nn.Module):
    def __init__(self, in_dim=3, hid=128, dropout=0.5, num_classes=5):
        super().__init__()
        self.dropout = dropout
        self.enc = nn.Linear(in_dim, hid)

        # GraphSAGE (mean aggregator)
        self.sage1 = SAGEConv(hid, hid)
        self.norm1 = nn.LayerNorm(hid)

        self.sage2 = SAGEConv(hid, hid)
        self.norm2 = nn.LayerNorm(hid)

        self.head = nn.Sequential(
            nn.Linear(hid * 2, hid),
            nn.ReLU(),
            nn.Dropout(dropout),
            nn.Linear(hid, num_classes)
        )

    def forward(self, data):
        x, edge_index, batch = data.x, data.edge_index, data.batch

        x = self.enc(x)
        x = F.relu(x)
        x = F.dropout(x, p=self.dropout, training=self.training)

        x = self.sage1(x, edge_index)
        x = self.norm1(x)
        x = F.relu(x)
        x = F.dropout(x, p=self.dropout, training=self.training)

        x = self.sage2(x, edge_index)
        x = self.norm2(x)
        x = F.relu(x)

        g_mean = global_mean_pool(x, batch)
        g_max = global_max_pool(x, batch)
        emb = torch.cat([g_mean, g_max], dim=1)

        logits = self.head(emb)
        return emb, logits


def train_encoder(model, train_loader, val_loader, device, class_weights_t,
                  epochs=60, lr=1e-3, ckpt_path="best_sage_focal.pt"):
    optimizer = torch.optim.AdamW(model.parameters(), lr=lr, weight_decay=1e-4)
    scheduler = torch.optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=epochs)
    criterion = FocalLoss(alpha=class_weights_t, gamma=3.0)

    best_val_acc = 0.0
    print("\n[Start GraphSAGE Training] (Focal Loss)")

    for ep in range(1, epochs + 1):
        model.train()
        total_loss = 0.0
        for data in train_loader:
            data = data.to(device)
            optimizer.zero_grad()
            _, logits = model(data)
            loss = criterion(logits, data.y)
            loss.backward()
            optimizer.step()
            total_loss += loss.item() * data.num_graphs

        scheduler.step()

        # Val acc
        model.eval()
        correct, total = 0, 0
        with torch.no_grad():
            for data in val_loader:
                data = data.to(device)
                _, logits = model(data)
                pred = logits.argmax(dim=1)
                correct += (pred == data.y).sum().item()
                total += data.y.size(0)

        val_acc = correct / max(total, 1)
        if val_acc > best_val_acc:
            best_val_acc = val_acc
            torch.save(model.state_dict(), ckpt_path)

        if ep % 10 == 0:
            print(f"Ep {ep:03d} | Loss {total_loss/max(len(train_loader.dataset),1):.4f} | Val Acc {val_acc:.4f} | Best {best_val_acc:.4f}")

    print(f"[SAGE Done] Best Val Acc: {best_val_acc:.4f}")


@torch.no_grad()
def extract_embeddings(model, loader, device):
    model.eval()
    embs, ys = [], []
    for data in loader:
        data = data.to(device)
        emb, _ = model(data)
        embs.append(emb.cpu().numpy())
        ys.append(data.y.cpu().numpy())
    return np.concatenate(embs, axis=0), np.concatenate(ys, axis=0)


# =========================
# 3) Main
# =========================
def main():
    seed_everything(42)
    PATH = r"adulteration_dataset_26_08_2021.csv"

    if not os.path.exists(PATH):
        print(f"File not found: {PATH}")
        return

    print(f"Loading data from {PATH}...")
    df = pd.read_csv(PATH)
    df["Concentration_Class"] = df["Concentration_Class"].astype(str)

    keep = {"0", "5", "10", "25", "50"}
    df = df[df["Concentration_Class"].isin(keep)].reset_index(drop=True)

    spec_cols = [c for c in df.columns if str(c).endswith("nm")]
    spec_cols = sorted(spec_cols, key=lambda x: float(str(x).replace("nm", "")))
    if len(spec_cols) == 0:
        raise ValueError("No spectral columns ending with 'nm' found.")

    X_raw = df[spec_cols].to_numpy(np.float32)   # [N,128]
    X = snv(X_raw)                               # [N,128]
    print(f"Input Shape: {X.shape} (No Downsampling)")

    classes_cc = ["0", "5", "10", "25", "50"]
    cc2id = {c: i for i, c in enumerate(classes_cc)}
    y = df["Concentration_Class"].map(cc2id).to_numpy(np.int64)

    if "Acquisition" not in df.columns:
        raise ValueError("Column 'Acquisition' is required for strict split.")
    acq = df["Acquisition"].to_numpy()

    train_idx = np.where(np.isin(acq, [1, 2, 3, 4]))[0]
    val_idx   = np.where(np.isin(acq, [5]))[0]
    test_idx  = np.where(np.isin(acq, [6]))[0]

    print("\n[Strict Split Info]")
    print(f"Train (Acq 1-4): {len(train_idx)}")
    print(f"Val   (Acq 5)  : {len(val_idx)}")
    print(f"Test  (Acq 6)  : {len(test_idx)}  <- FINAL BLIND")

    y_train_only = y[train_idx]
    class_w = compute_class_weights(y_train_only, num_classes=5, boost0=2.5)
    print(f"Class Weights: {np.round(class_w, 3)}")

    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    class_weights_t = torch.tensor(class_w, dtype=torch.float32, device=device)

    # Build graph (TRAIN only)
    EDGE_INDEX = build_hybrid_edge_index(X[train_idx], top_k=12)

    graphs_train = [spectrum_to_graph_sg3(X[i], int(y[i]), EDGE_INDEX) for i in train_idx]
    graphs_val   = [spectrum_to_graph_sg3(X[i], int(y[i]), EDGE_INDEX) for i in val_idx]
    graphs_test  = [spectrum_to_graph_sg3(X[i], int(y[i]), EDGE_INDEX) for i in test_idx]

    train_loader = DataLoader(graphs_train, batch_size=128, shuffle=True)
    val_loader   = DataLoader(graphs_val,   batch_size=256, shuffle=False)
    test_loader  = DataLoader(graphs_test,  batch_size=256, shuffle=False)

    # Train GraphSAGE
    ckpt = "best_sage_focal.pt"
    model = SAGEEncoder(in_dim=3, hid=128, dropout=0.5, num_classes=5).to(device)
    train_encoder(model, train_loader, val_loader, device, class_weights_t,
                  epochs=60, lr=1e-3, ckpt_path=ckpt)

    model.load_state_dict(torch.load(ckpt, map_location=device))

    # Extract embeddings
    Xtr_emb, ytr = extract_embeddings(model, train_loader, device)
    Xval_emb, yval = extract_embeddings(model, val_loader, device)
    Xte_emb, yte = extract_embeddings(model, test_loader, device)

    # LightGBM
    lgb_w = {i: float(class_w[i]) for i in range(5)}
    lgbm = lgb.LGBMClassifier(
        objective="multiclass",
        num_class=5,
        n_estimators=3000,
        learning_rate=0.01,
        num_leaves=31,
        max_depth=-1,
        min_child_samples=20,
        subsample=0.8,
        colsample_bytree=0.8,
        class_weight=lgb_w,
        random_state=42,
        n_jobs=-1,
        verbosity=-1
    )

    lgbm.fit(
        Xtr_emb, ytr,
        eval_set=[(Xval_emb, yval)],
        eval_metric="multi_logloss",
        callbacks=[lgb.early_stopping(stopping_rounds=100, verbose=True)]
    )

    best_iter = getattr(lgbm, "best_iteration_", None)
    if best_iter is None:
        best_iter = 3000
    print(f"[LGB] best_iteration_ = {best_iter}")

    # Refit on Train+Val
    X_trainval = np.vstack([Xtr_emb, Xval_emb])
    y_trainval = np.concatenate([ytr, yval])

    lgbm_final = lgb.LGBMClassifier(
        objective="multiclass",
        num_class=5,
        n_estimators=int(best_iter),
        learning_rate=0.01,
        num_leaves=31,
        max_depth=-1,
        min_child_samples=20,
        subsample=0.8,
        colsample_bytree=0.8,
        class_weight=lgb_w,
        random_state=42,
        n_jobs=-1,
        verbosity=-1
    )
    lgbm_final.fit(X_trainval, y_trainval)

    # Final blind test
    print("\n" + "=" * 46)
    print("[FINAL BLIND TEST] GraphSAGE + LGBM (Acq6)")
    print("=" * 46)

    pred = lgbm_final.predict(Xte_emb)
    bal = balanced_accuracy_score(yte, pred)
    mf1 = f1_score(yte, pred, average="macro")

    print(f"Balanced Acc: {bal:.4f}")
    print(f"Macro F1:     {mf1:.4f}")
    print("\nConfusion Matrix (rows=true, cols=pred):")
    print(confusion_matrix(yte, pred))
    print("\nPer-class report:")
    print(classification_report(yte, pred, target_names=classes_cc, digits=4))


if __name__ == "__main__":
    main()
